{{- /* Returns the page's 1200x630 Open Graph image, published at the path fk/og-name.html gives. The title is drawn on a transparent box (so long titles wrap inside the left area) and laid over og-base.png, which scripts/make_og_base.py renders. */ -}} {{- $p := . -}} {{- $isTR := eq $p.Language.Lang "tr" -}} {{- $title := $p.Params.ogTitle | default $p.Title | default site.Title -}} {{- $kicker := i18n "fk.og_tagline" -}} {{- if $p.IsHome -}} {{- $title = i18n "fk.og_home_title" -}} {{- $kicker = i18n "fk.status" -}} {{- else if $p.Params.project -}} {{- $kicker = printf "%s · %s" (i18n "fk.og_project") ($p.Params.categoryLabel | default "") -}} {{- else if $p.Params.ogKicker -}} {{- $kicker = $p.Params.ogKicker -}} {{- end -}} {{- $title = $title | truncate 64 | htmlUnescape -}} {{- /* Turkish casing rules: i -> İ, ı -> I. */ -}} {{- if $isTR }}{{ $kicker = replace (replace $kicker "i" "İ") "ı" "I" }}{{ end -}} {{- $kicker = upper $kicker -}} {{- $sub := "" -}} {{- with $p.Params.project }}{{ $sub = (index . $p.Language.Lang).summary | truncate 120 | htmlUnescape }}{{ end -}} {{- /* Font size and line count are estimated from the title length (Poppins Bold is about 0.6em per character on average, wider for lowercase words, the text box is 720px wide). A title without spaces cannot wrap, so it shrinks to fit one line. */ -}} {{- $n := strings.RuneCount $title -}} {{- $size := cond (gt $n 20) 52 60 -}} {{- if not (strings.Contains $title " ") }}{{ $size = math.Max 36 (math.Min 60 (math.Floor (div 700.0 (mul $n 0.7)))) | int }}{{ end -}} {{- $lines := math.Ceil (div (mul $n 0.6 $size) 720.0) | int -}} {{- $base := resources.Get "img/og-base.png" -}} {{- $box := resources.Get "img/og-textbox.png" -}} {{- $bold := resources.Get "fonts/Poppins-Bold.ttf" -}} {{- $mono := resources.Get "fonts/IBMPlexMono-Medium.ttf" -}} {{- $text := $box | images.Filter (images.Text $kicker (dict "font" $mono "size" 22 "color" "#5cc1ff" "x" 0 "y" 0)) (images.Text $title (dict "font" $bold "size" $size "color" "#f2f5f9" "linespacing" 6 "x" 0 "y" 46)) -}} {{- with $sub -}} {{- $regular := resources.Get "fonts/Poppins-Regular.ttf" -}} {{- $y := add 46 (mul $lines (add $size 6)) 20 -}} {{- $text = $text | images.Filter (images.Text . (dict "font" $regular "size" 26 "color" "#b9c5d6" "linespacing" 8 "x" 0 "y" $y)) -}} {{- end -}} {{- $img := $base | images.Filter (images.Overlay $text 80 150) -}} {{- $img = $img.Process "jpg q84" | resources.Copy (partial "fk/og-name.html" $p) -}} {{- return $img -}}
